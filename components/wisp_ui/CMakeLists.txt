# wisp_ui: the LVGL 9.5 runtime over wisp::hal::Display/Touch (see
# wisp/ui/ui_runtime.hpp), the device UI strings (wisp/ui/strings.hpp,
# wisp/ui/system_notice.hpp) and the system layer (wisp/ui/system_layer.hpp,
# implemented across src/system_layer.cpp, status_capsule.cpp,
# notification_panel.cpp and banner.cpp) and the wisp renderer
# (wisp/ui/wisp_view.hpp over wisp/ui/wisp_model.hpp, implemented across
# src/wisp_model.cpp, wisp_shape.cpp, wisp_spans.cpp and wisp_view.cpp). src/pinyin_dict.cpp is generated
# by tools/gen_pinyin_dict.py.
# fonts/lv_font_wisp_icons_{16,24,32,48}.c, the icons of the body and large
# fonts, and include/wisp/ui/symbols.hpp are generated by
# tools/gen_icon_font.py.
# lv_font_cjk_16.c is ~7MB of glyph data generated by tools/gen_cjk_font.py
# from Noto Sans CJK SC; expect the first build of this component to take a
# while. Only host builds compile it (WISP_UI_BUILTIN_CJK16), as the body
# font's fallback when no glyph image is given and as the reference the
# glyph image is tested against; the firmware reads the same font from
# the glyphs partition (glyph-partition spec 3.3).
#
# Self-contained include -- see cmake/wisp_component.cmake's header
# comment.
include(${CMAKE_CURRENT_LIST_DIR}/../../cmake/wisp_component.cmake)

# src/lv_mem_wisp.cpp holds LVGL's LV_STDLIB_CUSTOM heap hooks, which
# lvgl itself calls. On ESP-IDF it is part of wisp_ui (see the link note
# below). On the host `lvgl` is a static library that would then depend on
# wisp_ui and back, so simulator/CMakeLists.txt compiles the file into the
# `lvgl` target instead.
set(WISP_UI_SRCS
    src/a8_font.cpp src/align.cpp src/dialog.cpp src/font_image.cpp src/fonts.cpp src/glyph_image.cpp
    src/overlay.cpp src/theme.cpp src/ui_runtime.cpp src/strings.cpp src/strings_zh.cpp
    src/strings_ja.cpp src/strings_en.cpp src/system_layer.cpp src/status_capsule.cpp
    src/notification_panel.cpp src/banner.cpp src/system_notice.cpp src/pinyin_dict.cpp
    src/clock_text.cpp src/drag.cpp
    src/wisp_model.cpp src/wisp_shape.cpp src/wisp_spans.cpp src/wisp_view.cpp
    fonts/lv_font_wisp_icons_16.c fonts/lv_font_wisp_icons_24.c fonts/lv_font_wisp_icons_32.c
    fonts/lv_font_wisp_icons_48.c
)
if(ESP_PLATFORM)
    list(APPEND WISP_UI_SRCS src/lv_mem_wisp.cpp)
else()
    list(APPEND WISP_UI_SRCS fonts/lv_font_cjk_16.c)
endif()

wisp_component(
    NAME wisp_ui
    SRCS ${WISP_UI_SRCS}
    INCLUDE_DIRS include
    REQUIRES wisp_base wisp_os wisp_hal wisp_core
    IDF_REQUIRES lvgl__lvgl
    HOST_LIBS lvgl
)

if(NOT ESP_PLATFORM)
    target_compile_definitions(wisp_ui PRIVATE WISP_UI_BUILTIN_CJK16)
endif()

# Only lvgl calls the hooks in src/lv_mem_wisp.cpp, and the IDF link line
# can scan libwisp_ui.a for the last time before liblvgl__lvgl.a, so the
# linker is told up front that the object is needed (the same approach as
# wisp_port_esp's SIMD blend hooks). One symbol pulls in the whole object.
if(ESP_PLATFORM)
    target_link_libraries(${COMPONENT_LIB} INTERFACE "-u lv_malloc_core")
endif()

# Rendering path: always -O2 on the device, whatever the global level
# (connectivity spec 3.2; see firmware/CMakeLists.txt). The host build
# keeps its own flags.
if(ESP_PLATFORM)
    target_compile_options(${COMPONENT_LIB} PRIVATE -O2)
endif()
