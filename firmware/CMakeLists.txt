# ESP-IDF firmware project. Builds firmware/main (which pulls in
# wisp_app + wisp_port_esp, and -- via WISP_BOARD -- exactly one
# boards/<name>/board.cpp) against every portable wisp_* component plus
# the ESP-only wisp_port_esp port.
#
# Activate the pinned ESP-IDF v6.0.3 environment first (export.sh is a
# bash script; the user's shell is fish). Order: set-target -> bmgr ->
# build (see AGENTS.md section 4):
#   bash -c '. ~/esp/esp-idf-v6.0.3/export.sh && cd firmware && idf.py set-target esp32s3 && idf.py bmgr -c ../boards -b waveshare_amoled_216 && idf.py build'
# set-target selects the esp32s3 target and (via the component manager)
# fetches esp_board_manager, whose idf.py extension provides `idf.py bmgr`
# -- on a fresh checkout bmgr isn't available before that. bmgr then
# generates the gitignored components/gen_bmgr_codes/ from
# boards/<name>/board_*.yaml + setup_device.c; rerun it whenever the
# board YAML changes.

cmake_minimum_required(VERSION 3.16)

# Portable components (wisp_base/os/hal/core/ui/app) plus the ESP-only
# wisp_port_esp port both live under components/; board-specific code
# (selected below via WISP_BOARD) lives under drivers/ (per-chip
# drivers); EXTRA_COMPONENT_DIRS tolerates a path that doesn't exist, so
# a board that needs no extra chip drivers still builds without one.
set(EXTRA_COMPONENT_DIRS ${CMAKE_CURRENT_LIST_DIR}/../components ${CMAKE_CURRENT_LIST_DIR}/../drivers)

# Selects which boards/<name>/board.cpp firmware/main/CMakeLists.txt
# compiles into this binary (see wisp::board::create_board(), declared in
# components/wisp_app/include/wisp/app/board.hpp). Override on the
# command line (idf.py -DWISP_BOARD=<name> build) to build for a
# different board.
set(WISP_BOARD waveshare_amoled_216 CACHE STRING "Board to build for (boards/<name>/board.cpp)")

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# LVGL configuration (AGENTS.md's "LVGL 配置" trap): the lvgl/lvgl
# component's own CMake (env_support/cmake/esp.cmake) publicly defines
# LV_CONF_INCLUDE_SIMPLE, but lv_conf_internal.h checks LV_CONF_PATH
# *first* -- if it is defined (as it is here), that wins and LVGL
# #includes wisp's own components/wisp_ui/config/lv_conf.h instead of
# generating one from Kconfig. This must be set as a build-wide
# COMPILE_DEFINITIONS property (rather than e.g. a per-component
# target_compile_definitions call) between including project.cmake and
# calling project() below -- see that file's own header comment -- so it
# reaches every component's compile flags, including lvgl's own sources
# (idf_component_register() applies this property to each component it
# registers; see tools/cmake/component.cmake) and wisp_ui's
# ui_runtime.cpp, which both need to agree on the same lv_conf.h. The
# value must be the macro's literal replacement text, i.e. a
# double-quoted path (for `#include LV_CONF_PATH` to expand to
# `#include "<path>"`), not a bare path.
idf_build_set_property(
    COMPILE_DEFINITIONS
    "LV_CONF_PATH=\"${CMAKE_CURRENT_LIST_DIR}/../components/wisp_ui/config/lv_conf.h\""
    APPEND
)

project(wisp)

# esp_board_manager's generated component (components/gen_bmgr_codes/,
# regenerated by every `idf.py bmgr` run, so it cannot be edited) compiles
# the selected board directory via SRC_DIRS, i.e. every *.c/*.cpp in
# boards/<WISP_BOARD>/ -- which picks up Wisp's own board.cpp too. That
# file belongs to firmware/main (see main/CMakeLists.txt: it needs
# wisp_app/wisp_port_esp, which gen_bmgr_codes doesn't require, and
# compiling it twice would define create_board() twice), so drop it from
# gen_bmgr_codes' sources and leave it only setup_device.c (the bmgr
# factory entries).
if(TARGET __idf_gen_bmgr_codes)
    get_target_property(_wisp_bmgr_srcs __idf_gen_bmgr_codes SOURCES)
    list(FILTER _wisp_bmgr_srcs EXCLUDE REGEX "/boards/${WISP_BOARD}/board\\.cpp$")
    set_target_properties(__idf_gen_bmgr_codes PROPERTIES SOURCES "${_wisp_bmgr_srcs}")
endif()

# The admin web UI (connectivity spec 2.3): tools/pack_assets.py packs web/
# into build/assets.wpak, failing the build when it does not fit this
# board's `assets` partition or the gzip budget (spec 7.1), and `idf.py
# flash` writes it to that partition along with the app. The pack depends
# on every file and directory under web/: deleting or renaming a file only
# bumps its directory's mtime (same scheme as simulator/CMakeLists.txt).
set(WISP_WEB_DIR ${CMAKE_CURRENT_LIST_DIR}/../web)
set(WISP_ASSETS_WPAK ${CMAKE_BINARY_DIR}/assets.wpak)
file(GLOB_RECURSE WISP_WEB_FILES CONFIGURE_DEPENDS ${WISP_WEB_DIR}/*)
file(GLOB_RECURSE WISP_WEB_ENTRIES LIST_DIRECTORIES true CONFIGURE_DEPENDS ${WISP_WEB_DIR}/*)
list(APPEND WISP_WEB_FILES ${WISP_WEB_DIR})
foreach(entry IN LISTS WISP_WEB_ENTRIES)
    if(IS_DIRECTORY ${entry})
        list(APPEND WISP_WEB_FILES ${entry})
    endif()
endforeach()
idf_build_get_property(_wisp_python PYTHON)
set(WISP_BOARD_PARTITIONS ${CMAKE_CURRENT_LIST_DIR}/../boards/${WISP_BOARD}/partitions.csv)
add_custom_command(
    OUTPUT ${WISP_ASSETS_WPAK}
    COMMAND ${_wisp_python} ${CMAKE_CURRENT_LIST_DIR}/../tools/pack_assets.py ${WISP_WEB_DIR}
            -o ${WISP_ASSETS_WPAK} --partitions ${WISP_BOARD_PARTITIONS} --budget-kb 150
    DEPENDS ${CMAKE_CURRENT_LIST_DIR}/../tools/pack_assets.py ${WISP_BOARD_PARTITIONS}
            ${WISP_WEB_FILES}
    COMMENT "Packing web/ into assets.wpak"
    VERBATIM
)
add_custom_target(wisp_assets ALL DEPENDS ${WISP_ASSETS_WPAK})
esptool_py_flash_to_partition(flash assets ${WISP_ASSETS_WPAK})
add_dependencies(flash wisp_assets)
