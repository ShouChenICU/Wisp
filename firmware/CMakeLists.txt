# ESP-IDF firmware project. Builds firmware/main (which pulls in
# wisp_app + wisp_port_esp, and -- via WISP_BOARD -- exactly one
# boards/<name>/board.cpp) against every portable wisp_* component plus
# the ESP-only wisp_port_esp port.
#
# Activate the pinned ESP-IDF v6.0.3 environment first (export.sh is a
# bash script; the user's shell is fish). Order: set-target -> bmgr ->
# build (see AGENTS.md section 4):
#   bash -c '. ~/esp/esp-idf-v6.0.3/export.sh && cd firmware && idf.py set-target esp32s3 && idf.py bmgr -c ../boards -b waveshare_esp32_s3_touch_amoled_2_16 && idf.py build'
# set-target selects the esp32s3 target and (via the component manager)
# fetches esp_board_manager, whose idf.py extension provides `idf.py bmgr`
# -- on a fresh checkout bmgr isn't available before that. bmgr then
# generates the gitignored components/gen_bmgr_codes/ from
# boards/<name>/board_*.yaml + setup_device.c; rerun it whenever the
# board YAML changes.

cmake_minimum_required(VERSION 3.16)

# Portable components (wisp_base/os/hal/core/net/admin/ui/scene/agent/app) plus
# the ESP-only wisp_port_esp port all live under components/; vendored
# upstream sources (third_party/lua, used only by wisp_scene) under
# third_party/; board-specific code (selected below via WISP_BOARD) lives
# under drivers/ (per-chip drivers); EXTRA_COMPONENT_DIRS tolerates a path
# that doesn't exist, so a board that needs no extra chip drivers still
# builds without one.
set(EXTRA_COMPONENT_DIRS
    ${CMAKE_CURRENT_LIST_DIR}/../components
    ${CMAKE_CURRENT_LIST_DIR}/../third_party
    ${CMAKE_CURRENT_LIST_DIR}/../drivers
)

# Selects which boards/<name>/board.cpp firmware/main/CMakeLists.txt
# compiles into this binary (see wisp::board::create_board(), declared in
# components/wisp_app/include/wisp/app/board.hpp). Override on the
# command line (idf.py -DWISP_BOARD=<name> build) to build for a
# different board.
set(WISP_BOARD waveshare_esp32_s3_touch_amoled_2_16 CACHE STRING "Board to build for (boards/<name>/board.cpp)")

# The image's version (esp_app_desc_t::version, what the admin page and
# the firmware upload compare): X.Y.Z[-N-gHASH][-dirty] from git, see
# cmake/wisp_version.cmake. Must be set before project().
include(${CMAKE_CURRENT_LIST_DIR}/../cmake/wisp_version.cmake)
wisp_version(PROJECT_VER)

include($ENV{IDF_PATH}/tools/cmake/project.cmake)

# LVGL configuration (AGENTS.md's "LVGL 配置" trap): the lvgl/lvgl
# component's own CMake (env_support/cmake/esp.cmake) publicly defines
# LV_CONF_INCLUDE_SIMPLE, but lv_conf_internal.h checks LV_CONF_PATH
# *first* -- if it is defined (as it is here), that wins and LVGL
# #includes wisp's own components/wisp_ui/config/lv_conf.h instead of
# generating one from Kconfig. This must be set as a build-wide
# COMPILE_DEFINITIONS property (rather than e.g. a per-component
# target_compile_definitions call) between including project.cmake and
# calling project() below -- see that file's own header comment -- so it
# reaches every component's compile flags, including lvgl's own sources
# (idf_component_register() applies this property to each component it
# registers; see tools/cmake/component.cmake) and wisp_ui's
# ui_runtime.cpp, which both need to agree on the same lv_conf.h. The
# value must be the macro's literal replacement text, i.e. a
# double-quoted path (for `#include LV_CONF_PATH` to expand to
# `#include "<path>"`), not a bare path.
idf_build_set_property(
    COMPILE_DEFINITIONS
    "LV_CONF_PATH=\"${CMAKE_CURRENT_LIST_DIR}/../components/wisp_ui/config/lv_conf.h\""
    APPEND
)

# The ESP32-S3's SIMD fill and copy for LVGL's software renderer (see
# components/wisp_port_esp/lv_blend/esp32s3/lv_blend_esp32s3.h; the port
# component compiles the assembly for this target only). Build-wide for
# the same reason as LV_CONF_PATH: lv_conf.h reads it. Any other target
# leaves it undefined and builds LVGL's plain C path; a chip with its own
# draw unit or SIMD routines adds its own header here.
if(IDF_TARGET STREQUAL "esp32s3")
    idf_build_set_property(
        COMPILE_DEFINITIONS
        "WISP_LV_DRAW_SW_ASM_INCLUDE=\"${CMAKE_CURRENT_LIST_DIR}/../components/wisp_port_esp/lv_blend/esp32s3/lv_blend_esp32s3.h\""
        APPEND
    )
endif()

project(wisp)

# esp_board_manager's generated component (components/gen_bmgr_codes/,
# regenerated by every `idf.py bmgr` run, so it cannot be edited) compiles
# the selected board directory via SRC_DIRS, i.e. every *.c/*.cpp in
# boards/<WISP_BOARD>/ -- which picks up Wisp's own board.cpp too. That
# file belongs to firmware/main (see main/CMakeLists.txt: it needs
# wisp_app/wisp_port_esp, which gen_bmgr_codes doesn't require, and
# compiling it twice would define create_board() twice), so drop it from
# gen_bmgr_codes' sources and leave it only setup_device.c (the bmgr
# factory entries).
if(TARGET __idf_gen_bmgr_codes)
    get_target_property(_wisp_bmgr_srcs __idf_gen_bmgr_codes SOURCES)
    list(FILTER _wisp_bmgr_srcs EXCLUDE REGEX "/boards/${WISP_BOARD}/board\\.cpp$")
    set_target_properties(__idf_gen_bmgr_codes PROPERTIES SOURCES "${_wisp_bmgr_srcs}")
endif()

# The rendering path is always built with -O2, whatever the global
# optimization level (connectivity spec 3.2: the release build is -Os,
# the debug build -Og). IDF adds the global level as a directory-wide
# compile option, so a target's own option, which comes later on the
# command line, wins. wisp_ui and wisp_port_esp's display/touch sources
# set it themselves; LVGL and the esp_lcd* components (the panel and touch
# drivers and the LCD flush path) are third-party, so it is set here.
idf_build_get_property(_wisp_components BUILD_COMPONENTS)
foreach(_wisp_component IN LISTS _wisp_components)
    if(_wisp_component MATCHES "^lvgl__lvgl$|(^|__)esp_lcd")
        idf_component_get_property(_wisp_lib ${_wisp_component} COMPONENT_LIB)
        get_target_property(_wisp_type ${_wisp_lib} TYPE)
        if(NOT _wisp_type STREQUAL "INTERFACE_LIBRARY")
            target_compile_options(${_wisp_lib} PRIVATE -O2)
            list(APPEND _wisp_o2_components ${_wisp_component})
        endif()
    endif()
endforeach()
message(STATUS "Rendering path built with -O2: ${_wisp_o2_components}")
